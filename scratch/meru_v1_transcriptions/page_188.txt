ಕುಣ್ಡಸ್ಯ ಧೃತಿ-ಅಕ್ಷಪಾಟನಾದಿಸಂಸ್ಕಾರಾಃ

ಶ್ಲೋಕ ೨೬-೨೮:

ಸಮೀಕುರ್ಯಾತ್ತತೋಽಸ್ತ್ರೇಣ ಸೇಚಯೇತ್ಕವಚೇನ ತು ।
ಕಾಷ್ಠಾದಿನಾ ಕುಟ್ಟಯೇತ್ತನ್ಮನ್ತ್ರೇಣ ಸ್ಯಾತ್ತಥಾ ದೃಢಮ್ ॥೨೬॥
ಕುಶೈಃ ಸಮ್ಮಾರ್ಜಯೇತ್ಕುಣ್ಡಂ ತತಸ್ತು ಕವಚಾಣುನಾ ।
ವಿಲೇಪನಂ ಗೋಮಯಾದ್ಭಿಃ ಕವಚೇನೈವ ಕಾರಯೇತ್ ॥೨೭॥
ಕವಚೇ ನಾಗ್ನಿಸೂರ್ಯೋನ್ದುಕಲಾಃ ಕುಣ್ಡೇ ವಿಚಿನ್ತಯೇತ್ ।
ತ್ರಿಸೂತ್ರೀಕೃತಸೂತ್ರೇಣ ವೇಷ್ಟಯೇತ್ ಪ್ರತಿಮೇಖಲಮ್ ॥೨೮॥

ಶಬ್ದಾರ್ಥ:

• ಸಮೀಕುರ್ಯಾತ್ — ಸಮತಟ್ಟಾಗಿ ಮಾಡಬೇಕು (ಸಮೀಕರಣ ಸಂಸ್ಕಾರ) (ವಿಧಿಲಿಙ್ ಲಕಾರ)
• ತತಃ — ತರುವಾಯ (ಅವ್ಯಯ)
• ಅಸ್ತ್ರೇಣ — ಅಸ್ತ್ರಮಂತ್ರದಿಂದ (ತೃತೀಯಾ ಏಕವಚನ)
• ಸೇಚಯೇತ್ — ಜಲದಿಂದ ತೇವಗೊಳಿಸಬೇಕು (ಸೇಕ ಸಂಸ್ಕಾರ) (ವಿಧಿಲಿಙ್ ಲಕಾರ)
• ಕವಚೇನ — ಕವಚಮಂತ್ರದಿಂದ (ತೃತೀಯಾ ಏಕವಚನ)
• ತು — ಮತ್ತು (ಅವ್ಯಯ)
• ಕಾಷ್ಠಾದಿನಾ — ಯಜ್ಞಯೋಗ್ಯ ಮರದ ತುಂಡಿನಿಂದ (ತೃತೀಯಾ ಏಕವಚನ)
• ಕುಟ್ಟಯೇತ್ — ಗಟ್ಟಿಯಾಗಿ ಕುಟ್ಟಬೇಕು (ಕುಟ್ಟನ ಸಂಸ್ಕಾರ) (ವಿಧಿಲಿಙ್ ಲಕಾರ)
• ತನ್ಮನ್ತ್ರೇಣ — ಆಯಾ ಶಾಸ್ತ್ರೋಕ್ತ ಮಂತ್ರದಿಂದ (ತೃತೀಯಾ ಏಕವಚನ)
• ಸ್ಯಾತ್ — ಆಗುವಂತೆ (ವಿಧಿಲಿಙ್ ಲಕಾರ)
• ತಥಾ — ಆ ರೀತಿಯಾಗಿ (ಅವ್ಯಯ)
• ದೃಢಮ್ — ಅತ್ಯಂತ ಗಟ್ಟಿಯಾಗಿ, ದೃಢವಾಗಿ (ದ್ವಿತೀಯಾ ಏಕವಚನ)
• ಕುಶೈಃ — ದರ್ಭೆಗಳ ಕಂತೆಗಳಿಂದ (ತೃತೀಯಾ ಬಹುವಚನ)
• ಸಮ್ಮಾರ್ಜಯೇತ್ — ಸಾರಿಸಬೇಕು, ಶುದ್ಧಿಗೊಳಿಸಬೇಕು (ಸಮ್ಮಾರ್ಜನ ಸಂಸ್ಕಾರ) (ವಿಧಿಲಿಙ್ ಲಕಾರ)
• ಕುಣ್ಡಮ್ — ಯಾಗಕುಂಡವನ್ನು (ದ್ವಿತೀಯಾ ಏಕವಚನ)
• ತತಃ — ತರುವಾಯ (ಅವ್ಯಯ)
• ತು — ಆದರೆ (ಅವ್ಯಯ)
• ಕವಚಾಣುನಾ — ಕವಚಮಂತ್ರವಾದ ಹುಂ ಮಂತ್ರದಿಂದ (ತೃತೀಯಾ ಏಕವಚನ)
• ವಿಲೇಪನಂ — ಗೋಮಯ ಮತ್ತು ನೀರಿನಿಂದ ಸಾರಿಸುವುದು (ಲೇಪನ ಸಂಸ್ಕಾರ) (ದ್ವಿತೀಯಾ ಏಕವಚನ)
• ಗೋಮಯಾದ್ಭಿಃ — ಹಸುವಿನ ಸಗಣಿ ಮತ್ತು ಪವಿತ್ರ ಜಲದಿಂದ (ತೃತೀಯಾ ಬಹುವಚನ)
• ಕವಚೇನ — ಕವಚಮಂತ್ರದಿಂದ (ತೃತೀಯಾ ಏಕವಚನ)
• ಏವ — ಮಾತ್ರವೇ (ಅವ್ಯಯ)
• ಕಾರಯೇತ್ — ಮಾಡಿಸಬೇಕು (ವಿಧಿಲಿಙ್ ಲಕಾರ)
• ಕವಚೇನ — ಕವಚಮಂತ್ರದಿಂದ (ತೃತೀಯಾ ಏಕವಚನ)
• ಅಗ್ನಿಸೂರ್ಯೋನ್ದುಕಲಾಃ — ಅಗ್ನಿ, ಸೂರ್ಯ ಮತ್ತು ಚಂದ್ರರ ಸಮಸ್ತ ದಿವ್ಯ ಕಲೆಗಳನ್ನು (ದ್ವಿತೀಯಾ ಬಹುವಚನ)
• ಕುಣ್ಡೇ — ಕುಂಡದ ಒಳಭಾಗದಲ್ಲಿ (ಸಪ್ತಮೀ ಏಕವಚನ)
• ವಿಚಿನ್ತಯೇತ್ — ಧ್ಯಾನಿಸಬೇಕು (ಭಾವಿಸಬೇಕು) (ವಿಧಿಲಿಙ್ ಲಕಾರ)
• ತ್ರಿಸೂತ್ರೀಕೃತಸೂತ್ರೇಣ — ಮೂರು ಎಳೆಗಳನ್ನು ಒಟ್ಟಿಗೆ ಸೇರಿಸಿ ಹೊಸೆದ ರಕ್ಷಾಸೂತ್ರದಿಂದ (ತೃತೀಯಾ ಏಕವಚನ)
• ವೇಷ್ಟಯೇತ್ — ಸುತ್ತು ಕಟ್ಟಬೇಕು (ವೇಷ್ಟನ ಸಂಸ್ಕಾರ) (ವಿಧಿಲಿಙ್ ಲಕಾರ)
• ಪ್ರತಿಮೇಖಲಮ್ — ಕುಂಡದ ಪ್ರತಿಯೊಂದು ಮೇಖಲೆಗೂ (ಮೆಟ್ಟಿಲಿಗೂ) (ದ್ವಿತೀಯಾ ಏಕವಚನ)

ಭಾವಾರ್ಥ:

ಕುಂಡಸಂಸ್ಕಾರಗಳ ಮುಂದಿನ ಮಹತ್ವದ ಹಂತಗಳನ್ನು ಈ ಶ್ಲೋಕಗಳಲ್ಲಿ ವಿವರಿಸಲಾಗಿದೆ. ಅಗೆದು ತುಂಬಿದ ಮಣ್ಣನ್ನು ಅಸ್ತ್ರಮಂತ್ರದಿಂದ ಮೊದಲು ಸಮತಟ್ಟಾಗಿ ಮಾಡಬೇಕು (ಸಮೀಕರಣ ಸಂಸ್ಕಾರ). ನಂತರ ಕವಚಮಂತ್ರದಿಂದ ಜಲಪ್ರೋಕ್ಷಣೆ ಮಾಡಿ ಹದಗೊಳಿಸಬೇಕು (ಸೇಕ ಸಂಸ್ಕಾರ). ಯಜ್ಞಯೋಗ್ಯ ಮರದ ತುಂಡಿನಿಂದ (ಮುಸಲದಿಂದ) ಆ ಮಣ್ಣನ್ನು ಚೆನ್ನಾಗಿ ಕುಟ್ಟಿ, ಕುಂಡದ ಗೋಡೆಗಳು ಮತ್ತು ತಳವು ಕಲ್ಲಿನಂತೆ ಗಟ್ಟಿಯಾಗುವಂತೆ ಮಾಡಬೇಕು (ಕುಟ್ಟನ ಸಂಸ್ಕಾರ). ನಂತರ ಕವಚಮಂತ್ರದಿಂದ ದರ್ಭೆಗಳಿಂದ ಕುಂಡದ ಧೂಳನ್ನು ಗುಡಿಸಿ ಸ್ವಚ್ಛಗೊಳಿಸಬೇಕು (ಸಮ್ಮಾರ್ಜನ ಸಂಸ್ಕಾರ). ಹಸುವಿನ ಪವಿತ್ರ ಗೋಮಯ ಮತ್ತು ಗಂಗಾಜಲದಿಂದ ಕುಂಡದ ಸಮಸ್ತ ಭಾಗಗಳನ್ನೂ ಲೇಪನ ಮಾಡಬೇಕು (ಲೇಪನ ಸಂಸ್ಕಾರ). ತದನಂತರ ಕವಚಮಂತ್ರದಿಂದ ಕುಂಡದಲ್ಲಿ ಸೂರ್ಯನ ಹನ್ನೆರಡು ಕಲೆಗಳು, ಚಂದ್ರನ ಹದಿನಾರು ಕಲೆಗಳು ಮತ್ತು ಅಗ್ನಿಯ ಹತ್ತು ಕಲೆಗಳನ್ನು ಆವಾಹಿಸಿ ಧ್ಯಾನಿಸಬೇಕು. ಕೊನೆಯದಾಗಿ ಮೂರು ಎಳೆಗಳುಳ್ಳ ಪವಿತ್ರ ರಕ್ಷಾಸೂತ್ರದಿಂದ ಕುಂಡದ ಪ್ರತಿಯೊಂದು ಮೇಖಲೆಗೂ (ಮೆಟ್ಟಿಲಿಗೂ) ಪ್ರದಕ್ಷಿಣಾಕಾರವಾಗಿ ಸೂತ್ರವೇಷ್ಟನ ಮಾಡಬೇಕು (ವೇಷ್ಟನ ಸಂಸ್ಕಾರ). ಇದರಿಂದ ಕುಂಡವು ದೈವಿಕ ರಕ್ಷಾಕವಚದಿಂದ ಆವೃತವಾಗುತ್ತದೆ. ಮಣ್ಣನ್ನು ಕುಟ್ಟಿ ದೃಢಪಡಿಸುವುದು ಮತ್ತು ಸೂತ್ರವೇಷ್ಟನ ಮಾಡುವುದು ಕುಂಡಕ್ಕೆ ದೈವಿಕ ರಕ್ಷಾಕವಚವನ್ನು ನೀಡುತ್ತದೆ. ಸೂರ್ಯ, ಚಂದ್ರ ಮತ್ತು ಅಗ್ನಿಗಳ ತ್ರಿಕಲಾ ಶಕ್ತಿಯು ಕುಂಡದಲ್ಲಿ ಆವಾಹನೆಯಾದಾಗ ಅದು ತ್ರಿಲೋಕಗಳ ಪವಿತ್ರ ಸಂಗಮಸ್ಥಾನವಾಗುತ್ತದೆ. ಗೋಮಯಲೇಪನವು ಪರಿಸರವನ್ನು ಶುದ್ಧೀಕರಿಸಿ ಆಧ್ಯಾತ್ಮಿಕ ಕಂಪನಗಳನ್ನು ಹೆಚ್ಚಿಸುತ್ತದೆ.

ವಿಶೇಷ ವಿವರಣೆ / ವ್ಯಾಖ್ಯಾನ:

ಅನುವಾದ: ಅಸ್ತ್ರಮಂತ್ರದಿಂದ ಸಮೀಕರಣ, ಕವಚಮಂತ್ರದಿಂದ ಜಲಸೇಕ, ಮಂತ್ರಪೂರ್ವಕವಾಗಿ ಮರದಿಂದ ಕುಟ್ಟಿ ದೃಢೀಕರಣ, ಕವಚಮಂತ್ರದಿಂದ ಕುಶಗಳಿಂದ ಸಮ್ಮಾರ್ಜನ, ಗೋಮಯ-ಜಲದಿಂದ ವಿಲೇಪನ, ಅಗ್ನಿ-ಸೂರ್ಯ-ಚಂದ್ರ ಕಲೆಗಳ ಚಿಂತನೆ ಹಾಗೂ ಮೂರೆಳೆ ಸೂತ್ರದಿಂದ ಪ್ರತಿ ಮೇಖಲೆಗೂ ವೇಷ್ಟನ ಸಂಸ್ಕಾರವನ್ನು ಆಚರಿಸಬೇಕು.

ಅನುವಾದ:

ಅಸ್ತ್ರಮಂತ್ರದಿಂದ ಸಮತಟ್ಟು ಮಾಡಿ, ಕವಚದಿಂದ ನೀರೆರೆದು, ಮರದಿಂದ ಕುಟ್ಟಿ ಗಟ್ಟಿಗೊಳಿಸಿ, ದರ್ಭೆಯಿಂದ ಗುಡಿಸಿ, ಗೋಮಯದಿಂದ ಸಾರಿಸಿ, ಅಗ್ನಿ-ಸೂರ್ಯ-ಚಂದ್ರ ಕಲೆಗಳನ್ನು ಧ್ಯಾನಿಸಿ, ಮೂರೆಳೆ ಸೂತ್ರದಿಂದ ಪ್ರತಿಯೊಂದು ಮೇಖಲೆಗೂ ಸುತ್ತು ಕಟ್ಟಬೇಕು.

ಶ್ಲೋಕ ೨೯-೩೦:

ತ್ರಿವಾರಂ ವರ್ಮಮನ್ತ್ರೇಣ ಕುಣ್ಡಪೂಜಾಂ ಸಮಾಚರೇತ್ ।
ತಾರಶ್ಚಾಮುಕಕುಣ್ಡಾಯ ಆಸನಾಯ ನಮಸ್ತ್ವಿತಿ ॥೨೯॥
ಏಷೋಽರ್ಘ್ಯ ಇತ್ಯಾದ್ಯೂರ್ಹೈಶ್ಚ ಸಕಲಂ ಪೂಜನಂ ಚರೇತ್ ।
ನಾಭೇಶ್ಚ ಮೇಖಲಾನಾಂ ಚ ಕುರ್ಯಾದೇವಂ ಪ್ರಪೂಜನಮ್ ॥೩೦॥

ಶಬ್ದಾರ್ಥ:

• ತ್ರಿವಾರಂ — ಮೂರು ಬಾರಿ (ಅವ್ಯಯ)
• ವರ್ಮಮನ್ತ್ರೇಣ — ಕವಚಮಂತ್ರವಾದ ಹುಂ ಮಂತ್ರದಿಂದ (ತೃತೀಯಾ ಏಕವಚನ)
• ಕುಣ್ಡಪೂಜಾಂ — ಕುಂಡದ ಷೋಡಶೋಪಚಾರ ಪೂಜೆಯನ್ನು (ದ್ವಿತೀಯಾ ಏಕವಚನ)
• ಸಮಾಚರೇತ್ — ವಿಧಿವತ್ತಾಗಿ ನೆರವೇರಿಸಬೇಕು (ವಿಧಿಲಿಙ್ ಲಕಾರ)
• ತಾರಃ — ಓಂಕಾರ ಪ್ರಣವ ಮಂತ್ರವು (ಪುಲ್ಲಿಂಗ, ಪ್ರಥಮಾ ಏಕವಚನ)
• ಚ — ಮತ್ತು (ಅವ್ಯಯ)
• ಅಮುಕಕುಣ್ಡಾಯ — ಆಯಾ ನಿರ್ದಿಷ್ಟ ಕುಂಡದ ಹೆಸರನ್ನು ಹೇಳಿ (ಚತುರ್ಥೀ ಏಕವಚನ)
• ಆಸನಾಯ — ಆಸನ ಸಮರ್ಪಣೆಗಾಗಿ (ಚತುರ್ಥೀ ಏಕವಚನ)
• ನಮಃ ತು ಇತಿ — ನಮಸ್ಕಾರವೆಂದು ಮಂತ್ರೋಚ್ಚಾರಣೆ ಮಾಡುತ್ತಾ (ಅವ್ಯಯ)
• ಏಷಃ ಅರ್ಘ್ಯಃ — ಇದು ಅರ್ಘ್ಯವು (ಪ್ರಥಮಾ ಏಕವಚನ)
• ಇತ್ಯಾದಿ ಊಹೈಃ — ಮುಂತಾದ ಉಪಚಾರ ವಾಕ್ಯಗಳ ಕಲ್ಪನೆಯಿಂದ (ತೃತೀಯಾ ಬಹುವಚನ)
• ಚ — ಮತ್ತು (ಅವ್ಯಯ)
• ಸಕಲಂ — ಸಮಸ್ತವಾದ (ವಿಶೇಷಣ)
• ಪೂಜನಂ — ಪೂಜಾವಿಧಾನವನ್ನು (ದ್ವಿತೀಯಾ ಏಕವಚನ)
• ಚರೇತ್ — ಆಚರಿಸಬೇಕು (ವಿಧಿಲಿಙ್ ಲಕಾರ)
• ನಾಭೇಃ — ಕುಂಡದ ಮಧ್ಯದಲ್ಲಿರುವ ಕಮಲಾಕಾರದ ನಾಭಿಯ (ಷಷ್ಠೀ ಏಕವಚನ)
• ಚ — ಮತ್ತು (ಅವ್ಯಯ)
• ಮೇಖಲಾನಾಂ — ಸುತ್ತಲಿರುವ ಮೂರು ಮೇಖಲೆಗಳ (ಷಷ್ಠೀ ಬಹುವಚನ)
• ಚ — ಮತ್ತು (ಅವ್ಯಯ)
• ಕುರ್ಯಾತ್ — ಮಾಡಬೇಕು (ವಿಧಿಲಿಙ್ ಲಕಾರ)
• ಏವಂ — ಈ ಪ್ರಕಾರವಾಗಿ (ಅವ್ಯಯ)
• ಪ್ರಪೂಜನಮ್ — ಭಕ್ತಿಪೂರ್ವಕ ಆರಾಧನೆಯನ್ನು (ದ್ವಿತೀಯಾ ಏಕವಚನ)

ಭಾವಾರ್ಥ:

ಕುಂಡದ ಪ್ರತ್ಯಕ್ಷ ಪೂಜಾ ವಿಧಾನವನ್ನು ಈ ಶ್ಲೋಕಗಳಲ್ಲಿ ನಿರೂಪಿಸಲಾಗಿದೆ. ಕವಚಮಂತ್ರವನ್ನು ಮೂರು ಬಾರಿ ಉಚ್ಚರಿಸುತ್ತಾ ಕುಂಡಕ್ಕೆ ಸಾಕ್ಷಾತ್ ದೇವತಾಭಾವದಿಂದ ಪೂಜೆಯನ್ನು ಸಲ್ಲಿಸಬೇಕು. ಪ್ರಣವವಾದ 'ಓಂ' ಕಾರವನ್ನು ಮೊದಲು ಸೇರಿಸಿ, ಆಯಾ ಕುಂಡದ ಹೆಸರನ್ನು ಉಚ್ಚರಿಸಿ — ಉದಾಹರಣೆಗೆ 'ಓಂ ಚತುರಸ್ರಕುಂಡಾಯ ಆಸನಾಯ ನಮಃ' ಎಂದು ಆಸನವನ್ನು ಸಮರ್ಪಿಸಬೇಕು. ತದನಂತರ ಅರ್ಘ್ಯ, ಪಾದ್ಯ, ಆಚಮನೀಯ, ಗಂಧ, ಅಕ್ಷತೆ, ಪುಷ್ಪ, ಧೂಪ, ದೀಪ, ನೈವೇದ್ಯ, ತಾಂಬೂಲ ಮುಂತಾದ ಸಮಸ್ತ ಷೋಡಶೋಪಚಾರಗಳನ್ನು ಶಾಸ್ತ್ರೋಕ್ತವಾಗಿ ಅರ್ಪಿಸಬೇಕು. ಕೇವಲ ಕುಂಡದ ಒಳಭಾಗವನ್ನು ಮಾತ್ರವಲ್ಲದೆ, ತಳದಲ್ಲಿರುವ ಪದ್ಮನಾಭಿಯನ್ನೂ ಮತ್ತು ಸುತ್ತಲಿರುವ ಮೂರೂ ಮೇಖಲೆಗಳನ್ನೂ ದೈವಿಕ ಶಕ್ತಿಗಳ ಆವಾಸಸ್ಥಾನವೆಂದು ಭಾವಿಸಿ ಪ್ರತ್ಯೇಕವಾಗಿ ಪೂಜಿಸಬೇಕು. ಇದರಿಂದ ಕುಂಡವು ಸಕಲ ದೇವತೆಗಳ ಸನ್ನಿಧಾನವಾಗಿ ಪರಿವರ್ತಿತವಾಗುತ್ತದೆ. ಕುಂಡವನ್ನು ಸಾಕ್ಷಾತ್ ದೇವತೆಯೆಂದು ಭಾವಿಸಿ ಪೂಜಿಸುವುದರಿಂದ ಅಗ್ನಿಹೋಮವು ಪರಮ ಪವಿತ್ರ ದೈವಿಕ ಸಮರ್ಪಣೆಯಾಗುತ್ತದೆ. ನೈವೇದ್ಯ, ಧೂಪ, ದೀಪಾದಿ ಉಪಚಾರಗಳು ಕುಂಡಾಧಿಷ್ಠಿತ ದೇವತೆಗಳನ್ನು ಸಂತುಷ್ಟಗೊಳಿಸಿ ಅನುಗ್ರಹವನ್ನು ಆಕರ್ಷಿಸುತ್ತವೆ. ಶ್ರದ್ಧಾಪೂರ್ವಕ ಪೂಜೆಯಿಲ್ಲದೆ ಮಾಡುವ ಹೋಮವು ಎಂದಿಗೂ ಸಿದ್ಧಿಯನ್ನು ತರಲಾರದು. ಇಂತಹ ಶ್ರದ್ಧಾಭಕ್ತಿಗಳ ಸಮರ್ಪಣೆಯು ಕುಂಡದಲ್ಲಿ ಸಾಕ್ಷಾತ್ ದೈವಿಕ ಸನ್ನಿಧಾನವನ್ನು ಪ್ರತ್ಯಕ್ಷವಾಗಿ ನೆಲೆಗೊಳಿಸುತ್ತದೆ.

ವಿಶೇಷ ವಿವರಣೆ / ವ್ಯಾಖ್ಯಾನ:

ಅನುವಾದ: ಕವಚಮಂತ್ರದಿಂದ ಮೂರು ಬಾರಿ ಕುಂಡದ ಪೂಜೆ ಮಾಡಬೇಕು. 'ಓಂ ಅಮುಕಕುಂಡಾಯ ಆಸನಾಯ ನಮಃ' ಎಂದು ಆಸನ ಕೊಟ್ಟು, ನಂತರ ಅರ್ಘ್ಯ, ಪಾದ್ಯ, ಗಂಧ, ಪುಷ್ಪ ಮುಂತಾದ ಸಕಲ ಉಪಚಾರಗಳಿಂದ ಕುಂಡ, ನಾಭಿ ಮತ್ತು ಮೇಖಲೆಗಳನ್ನು ಪೂಜಿಸಬೇಕು.

ಅನುವಾದ:

ಮೂರು ಬಾರಿ ಕವಚಮಂತ್ರದಿಂದ ಕುಂಡವನ್ನು ಪೂಜಿಸಬೇಕು. ಓಂಕಾರಪೂರ್ವಕವಾಗಿ ಆಯಾ ಕುಂಡದ ಹೆಸರಿನಿಂದ ಆಸನ, ಅರ್ಘ್ಯಾದಿ ಸಮಸ್ತ ಉಪಚಾರಗಳನ್ನು ಸಮರ್ಪಿಸಿ, ನಾಭಿ ಮತ್ತು ಮೇಖಲೆಗಳನ್ನೂ ವಿಧಿವತ್ತಾಗಿ ಪೂಜಿಸಬೇಕು.

ಶ್ಲೋಕ ೩೧-೩೩:

ಸಞ್ಚಿನ್ತಯೇತ್ತತೋಽಸ್ತ್ರೇಣ ತತ್ಕುಣ್ಡಂ ವಜ್ರವದ್ದೃಢಮ್ ।
ವಹ್ನೇರ್ಜ್ವಾಲಾನಿರಾಸಾಯ ಹೃನ್ಮನ್ತ್ರೇಣ ಕುಶೈಸ್ತತಃ ॥೩೧॥
ಕುಣ್ಡಸ್ಯ ತು ಚತುರ್ದಿಕ್ಕು ಕುರ್ಯಾನ್ಮಾರ್ಗಚತುಷ್ಟಯಮ್ ।
ಅಚ್ಛಿನ್ನಾಗ್ರೈಃ ಕುಶೈರಸ್ತ್ರಮನ್ತ್ರಿತೈಃ ಕುಣ್ಡಭಿತ್ತಿಷು ॥೩೨॥
ಭೂತಪ್ರೇತಾದಿದೃಷ್ಟೀನಾಮಸ್ತ್ರಾದೀನಾಂ ಸಮಾಚರೇತ್ ।
ಉಚ್ಚಾಟನಮಿಮೇ ಪ್ರೋಕ್ತಾಃ ಸಂಸ್ಕಾರಾ ಧೃತಿಸಮ್ಮಿತಾಃ ॥೩೩॥

ಶಬ್ದಾರ್ಥ:

• ಸಞ್ಚಿನ್ತಯೇತ್ — ಮನಸ್ಸಿನಲ್ಲಿ ದೃಢವಾಗಿ ಭಾವಿಸಬೇಕು (ವಿಧಿಲಿಙ್ ಲಕಾರ)
• ತತಃ — ಆ ನಂತರ (ಅವ್ಯಯ)
• ಅಸ್ತ್ರೇಣ — ಅಸ್ತ್ರಮಂತ್ರದಿಂದ (ತೃತೀಯಾ ಏಕವಚನ)
• ತತ್ — ಆ ಯಜ್ಞಕುಂಡವನ್ನು (ಸರ್ವನಾಮ)
• ಕುಣ್ಡಮ್ — ಕುಂಡವನ್ನು (ದ್ವಿತೀಯಾ ಏಕವಚನ)
• ವಜ್ರವತ್ — ಇಂದ್ರನ ವಜ್ರಾಯುಧದಂತೆ ಅಭೇದ್ಯವಾಗಿ (ಅವ್ಯಯ)
• ದೃಢಮ್ — ಅತ್ಯಂತ ಗಟ್ಟಿಯಾಗಿದೆ ಎಂದು (ದ್ವಿತೀಯಾ ಏಕವಚನ)
• ವಹ್ನೇಃ — ಪ್ರಜ್ವಲಿಸುವ ಅಗ್ನಿಯ (ಷಷ್ಠೀ ಏಕವಚನ)
• ಜ್ವಾಲಾನಿರಾಸಾಯ — ಅಗ್ನಿಜ್ವಾಲೆಯು ಅತಿಯಾಗಿ ಹೊರಚಿಮ್ಮುವುದನ್ನು ತಡೆಯಲು (ಚತುರ್ಥೀ ಏಕವಚನ)
• ಹೃನ್ಮನ್ತ್ರೇಣ — ನಮಃ ಎಂಬ ಹೃದಯಮಂತ್ರದಿಂದ (ತೃತೀಯಾ ಏಕವಚನ)
• ಕುಶೈಃ — ದರ್ಭೆಗಳಿಂದ (ತೃತೀಯಾ ಬಹುವಚನ)
• ತತಃ — ತರುವಾಯ (ಅವ್ಯಯ)
• ಕುಣ್ಡಸ್ಯ — ಕುಂಡದ (ಷಷ್ಠೀ ಏಕವಚನ)
• ತು — ಆದರೆ (ಅವ್ಯಯ)
• ಚತುರ್ದಿಕ್ಕು — ನಾಲ್ಕೂ ದಿಕ್ಕುಗಳಲ್ಲಿಯೂ (ಸಪ್ತಮೀ ಬಹುವಚನ)
• ಕುರ್ಯಾತ್ — ನಿರ್ಮಿಸಬೇಕು (ವಿಧಿಲಿಙ್ ಲಕಾರ)
• ಮಾರ್ಗಚತುಷ್ಟಯಮ್ — ನಾಲ್ಕು ಪವಿತ್ರ ಜ್ವಾಲಾ ನಿರ್ಗಮನ ಮಾರ್ಗಗಳನ್ನು (ದ್ವಿತೀಯಾ ಏಕವಚನ)
• ಅಚ್ಛಿನ್ನಾಗ್ರೈಃ — ತುದಿ ಮುರಿಯದ ಅಖಂಡವಾದ (ತೃತೀಯಾ ಬಹುವಚನ)
• ಕುಶೈಃ — ದರ್ಭೆಗಳಿಂದ (ತೃತೀಯಾ ಬಹುವಚನ)
• ಅಸ್ತ್ರಮನ್ತ್ರಿತೈಃ — ಅಸ್ತ್ರಮಂತ್ರದಿಂದ ಅಭಿಮಂತ್ರಿಸಲ್ಪಟ್ಟ (ತೃತೀಯಾ ಬಹುವಚನ)
• ಕುಣ್ಡಭಿತ್ತಿಷು — ಕುಂಡದ ನಾಲ್ಕೂ ಗೋಡೆಗಳ ಮೇಲೆ (ಸಪ್ತಮೀ ಬಹುವಚನ)
• ಭೂತಪ್ರೇತಾದಿದೃಷ್ಟೀನಾಮ್ — ದುಷ್ಟ ಭೂತ, ಪ್ರೇತ, ರಾಕ್ಷಸರ ಕೆಟ್ಟ ದೃಷ್ಟಿಗಳನ್ನು (ಷಷ್ಠೀ ಬಹುವಚನ)
• ಅಸ್ತ್ರಾದೀನಾಮ್ — ಶತ್ರುಗಳ ಮಾಟ-ಮಂತ್ರಗಳ ದುಷ್ಪ್ರಭಾವಗಳನ್ನು (ಷಷ್ಠೀ ಬಹುವಚನ)
• ಸಮಾಚರೇತ್ — ಆಚರಿಸಬೇಕು (ವಿಧಿಲಿಙ್ ಲಕಾರ)
• ಉಚ್ಚಾಟನಮ್ — ಸಂಪೂರ್ಣವಾಗಿ ಓಡಿಸಿ ನಾಶಪಡಿಸುವುದನ್ನು (ದ್ವಿತೀಯಾ ಏಕವಚನ)
• ಇಮೇ — ಈ ಸಮಸ್ತ ವಿಧಿಗಳು (ಪ್ರಥಮಾ ಬಹುವಚನ)
• ಪ್ರೋಕ್ತಾಃ — ಹೇಳಲ್ಪಟ್ಟಿವೆ (ಪ್ರಥಮಾ ಬಹುವಚನ)
• ಸಂಸ್ಕಾರಾಃ — ಪವಿತ್ರ ಸಂಸ್ಕಾರಗಳೆಂದು (ಪ್ರಥಮಾ ಬಹುವಚನ)
• ಧೃತಿಸಮ್ಮಿತಾಃ — ಧೃತಿ ಸಂಸ್ಕಾರದ ಸಮಾನವೆಂದು ಶಾಸ್ತ್ರಜ್ಞರಿಂದ (ಪ್ರಥಮಾ ಬಹುವಚನ)

ಭಾವಾರ್ಥ:

ಕುಂಡದ ರಕ್ಷಣೆ ಮತ್ತು ದೈವಿಕ ಸ್ಥಿರತೆಗಾಗಿ ನಡೆಸಲಾಗುವ 'ಧೃತಿ ಸಂಸ್ಕಾರ'ದ ಅದ್ಭುತ ರಹಸ್ಯವನ್ನು ಈ ಶ್ಲೋಕಗಳಲ್ಲಿ ನಿರೂಪಿಸಲಾಗಿದೆ. ಸಾಧಕನು ಅಸ್ತ್ರಮಂತ್ರವನ್ನು ಪಠಿಸುತ್ತಾ, ಈ ಯಜ್ಞಕುಂಡವು ದೇವೇಂದ್ರನ ವಜ್ರಾಯುಧದಂತೆ ಅತ್ಯಂತ ಅಭೇದ್ಯವೂ ದೃಢವೂ ಆಗಿದೆ ಎಂದು ಭಾವಿಸಬೇಕು (ವಜ್ರೀಕರಣ ಸಂಸ್ಕಾರ). ನಂತರ ಹೋಮದ ಸಮಯದಲ್ಲಿ ಅಗ್ನಿಯ ಪ್ರಚಂಡ ಜ್ವಾಲೆಯು ನಿಯಂತ್ರಣ ತಪ್ಪಿ ಹೊರಚಿಮ್ಮದಂತೆ ಮಾಡಲು, ಹೃದಯಮಂತ್ರವನ್ನು (ನಮಃ) ಜಪಿಸುತ್ತಾ ದರ್ಭೆಗಳಿಂದ ಕುಂಡದ ನಾಲ್ಕೂ ದಿಕ್ಕುಗಳಲ್ಲಿ ನಾಲ್ಕು ನಿರ್ಗಮನ ಮಾರ್ಗಗಳನ್ನು ರೂಪಿಸಬೇಕು. ತದನಂತರ ತುದಿ ಮುರಿಯದ ಅಖಂಡ ದರ್ಭೆಗಳನ್ನು ಅಸ್ತ್ರಮಂತ್ರದಿಂದ ಅಭಿಮಂತ್ರಿಸಿ, ಕುಂಡದ ಗೋಡೆಗಳ ಮೇಲೆ ರೇಖೆಗಳನ್ನು ಎಳೆಯುತ್ತಾ, ಜೋರಾದ ಧ್ವನಿಯಲ್ಲಿ ಅಸ್ತ್ರಮಂತ್ರವನ್ನು ಪಠಿಸಿ ಯಜ್ಞವನ್ನು ಕೆಡಿಸಲು ಹವಣಿಸುವ ಸಮಸ್ತ ದುಷ್ಟ ಭೂತ, ಪ್ರೇತ, ಪಿಶಾಚ ಹಾಗೂ ಶತ್ರುಗಳ ಮಾಟ-ಮಂತ್ರಗಳ ದೃಷ್ಟಿದೋಷಗಳನ್ನು ಉಚ್ಚಾಟನೆ ಮಾಡಬೇಕು. ಈ ಸಮಸ್ತ ರಕ್ಷಣಾತ್ಮಕ ಪ್ರಕ್ರಿಯೆಗಳಿಗೆ ತಂತ್ರಶಾಸ್ತ್ರದಲ್ಲಿ 'ಧೃತಿ ಸಂಸ್ಕಾರ' ಎಂದು ಕರೆಯಲಾಗುತ್ತದೆ. ವಜ್ರೀಕರಣ ಮತ್ತು ದೃಷ್ಟಿದೋಷ ನಿವಾರಣೆಗಳು ಯಜ್ಞಕ್ಷೇತ್ರವನ್ನು ಸಮಸ್ತ ನಕಾರಾತ್ಮಕ ಶಕ್ತಿಗಳಿಂದ ಸಂಪೂರ್ಣವಾಗಿ ರಕ್ಷಿಸುತ್ತವೆ. ಅಗ್ನಿಯ ಪ್ರಚಂಡ ಜ್ವಾಲೆಯು ಸಮತೋಲನದಲ್ಲಿ ಪ್ರಜ್ವಲಿಸಲು ಈ ಶಾಸ್ತ್ರೀಯ ಮಾರ್ಗಗಳು ಅತ್ಯಂತ ವೈಜ್ಞಾನಿಕವಾಗಿ ನೆರವಾಗುತ್ತವೆ. ಧೃತಿ ಸಂಸ್ಕಾರವು ಯಜ್ಞದ ನಿರ್ವಿಘ್ನ ಸಮಾಪ್ತಿಗೆ ಭದ್ರ ಬುನಾದಿಯನ್ನು ಒದಗಿಸುತ್ತದೆ.

ವಿಶೇಷ ವಿವರಣೆ / ವ್ಯಾಖ್ಯಾನ:

ಅನುವಾದ: ಅಸ್ತ್ರಮಂತ್ರದಿಂದ ಕುಂಡವು ವಜ್ರದಂತೆ ದೃಢವಾಗಿದೆ ಎಂದು ಭಾವಿಸಬೇಕು. ಅಗ್ನಿಜ್ವಾಲೆಯ ಶಾಂತಿಗಾಗಿ ಹೃದಯಮಂತ್ರದಿಂದ ಕುಶಗಳಿಂದ ನಾಲ್ಕೂ ದಿಕ್ಕುಗಳಲ್ಲಿ ನಾಲ್ಕು ಮಾರ್ಗಗಳನ್ನು ಮಾಡಬೇಕು. ತುದಿ ಮುರಿಯದ ಕುಶಗಳಿಂದ ಅಸ್ತ್ರಮಂತ್ರವನ್ನು ಪಠಿಸುತ್ತಾ ಕುಂಡದ ಗೋಡೆಗಳ ಮೇಲೆ ರೇಖೆಗಳನ್ನು ಎಳೆದು, ಭೂತ-ಪ್ರೇತಾದಿಗಳ ದುಷ್ಟದೃಷ್ಟಿಯನ್ನು ಉಚ್ಚಾಟನೆ ಮಾಡಬೇಕು. ಇವು ಧೃತಿ ಸಂಸ್ಕಾರಗಳೆನಿಸುತ್ತವೆ.

ಅನುವಾದ:

ಅಸ್ತ್ರಮಂತ್ರದಿಂದ ಕುಂಡವನ್ನು ವಜ್ರದಂತೆ ದೃಢವೆಂದು ಭಾವಿಸಿ, ಹೃದಯಮಂತ್ರದಿಂದ ಜ್ವಾಲಾನಿಯಂತ್ರಣಕ್ಕೆ ನಾಲ್ಕು ಮಾರ್ಗಗಳನ್ನು ಮಾಡಿ, ಅಖಂಡ ಕುಶಗಳಿಂದ ಅಸ್ತ್ರಮಂತ್ರೋಚ್ಚಾರಣೆ ಮಾಡುತ್ತಾ ಭೂತಪ್ರೇತಾದಿಗಳ ದೃಷ್ಟಿಗಳನ್ನು ಉಚ್ಚಾಟಿಸಬೇಕು. ಇದು ಧೃತಿ ಸಂಸ್ಕಾರವೆನಿಸುತ್ತದೆ.

ಶ್ಲೋಕ ೩೪:

ಭವೇದನೇನ ವಿಧಿನಾ ಕುಣ್ಡಸ್ಯಾಪ್ಯಕ್ಷಪಾಟನಮ್ ।
ವಾಮಹೋಮೇ ತು ಸಂಸ್ಕಾರಾ ಆದ್ಯಾಶ್ಚತ್ವಾರ ಏವ ಹಿ ॥೩೪॥

ಶಬ್ದಾರ್ಥ:

• ಭವೇತ್ — ನೆರವೇರುತ್ತದೆ (ವಿಧಿಲಿಙ್ ಲಕಾರ)
• ಅನೇನ — ಈ ಮೇಲೆ ವಿವರಿಸಿದ (ತೃತೀಯಾ ಏಕವಚನ)
• ವಿಧಿನಾ — ಶಾಸ್ತ್ರೀಯ ವಿಧಿವಿಧಾನದಿಂದ (ತೃತೀಯಾ ಏಕವಚನ)
• ಕುಣ್ಡಸ್ಯ — ಯಾಗಕುಂಡದ (ಷಷ್ಠೀ ಏಕವಚನ)
• ಅಪಿ — ಸಹ (ಅವ್ಯಯ)
• ಅಕ್ಷಪಾಟನಮ್ — ಕಣ್ಣುಗಳನ್ನು ತೆರೆಸುವ ಅಕ್ಷಪಾಟನ ಸಂಸ್ಕಾರವು (ನಪುಂಸಕಲಿಂಗ, ಪ್ರಥಮಾ ಏಕವಚನ)
• ವಾಮಹೋಮೇ — ತಂತ್ರಶಾಸ್ತ್ರದ ವಾಮಮಾರ್ಗದ ಹೋಮಾನುಷ್ಠಾನದಲ್ಲಿ (ಸಪ್ತಮೀ ಏಕವಚನ)
• ತು — ಆದರೆ (ಅವ್ಯಯ)
• ಸಂಸ್ಕಾರಾಃ — ಕುಂಡಸಂಸ್ಕಾರಗಳು (ಪುಲ್ಲಿಂಗ, ಪ್ರಥಮಾ ಬಹುವಚನ)
• ಆದ್ಯಾಃ — ಪ್ರಾರಂಭದ (ಪ್ರಥಮಾ ಬಹುವಚನ)
• ಚತ್ವಾರಃ — ಕೇವಲ ನಾಲ್ಕು ಸಂಸ್ಕಾರಗಳು (ಸಂಖ್ಯಾವಾಚಕ)
• ಏವ — ಮಾತ್ರವೇ (ಅವ್ಯಯ)
• ಹಿ — ನಿಶ್ಚಯವಾಗಿಯೂ (ಅವ್ಯಯ)

ಭಾವಾರ್ಥ:

ಕುಂಡದ ದೃಷ್ಟಿಪ್ರದಾನ ಸಂಸ್ಕಾರ ಹಾಗೂ ವಾಮಮಾರ್ಗದ ವಿಶಿಷ್ಟ ನಿಯಮವನ್ನು ಈ ಶ್ಲೋಕದಲ್ಲಿ ಸ್ಪಷ್ಟಪಡಿಸಲಾಗಿದೆ. ಮೇಲ್ಕಂಡ ಸಮಸ್ತ ವಿಧಿಗಳನ್ನು ಕ್ರಮಬದ್ಧವಾಗಿ ಆಚರಿಸಿದಾಗ, ಜಡವಾಗಿದ್ದ ಕುಂಡಕ್ಕೆ ದೈವಿಕ ದೃಷ್ಟಿ ಪ್ರಾಪ್ತವಾಗುವ 'ಅಕ್ಷಪಾಟನ ಸಂಸ್ಕಾರ'ವು ಸಂಪೂರ್ಣವಾಗುತ್ತದೆ. ಅಂದರೆ ಕುಂಡವು ಅಗ್ನಿಯ ತೇಜಸ್ಸನ್ನು ಸ್ವೀಕರಿಸಲು ಜೀವಂತ ಚೈತನ್ಯಮಯ ಕ್ಷೇತ್ರವಾಗಿ ಜಾಗೃತಗೊಳ್ಳುತ್ತದೆ. ಆದರೆ ದಕ್ಷಿಣಮಾರ್ಗದಲ್ಲಿ ಈ ಹದಿನೆಂಟು ಸಂಸ್ಕಾರಗಳೂ ಅನಿವಾರ್ಯವಾಗಿದ್ದರೆ, ತಂತ್ರದ ವಾಮಮಾರ್ಗದ ಹೋಮದಲ್ಲಿ ಇಷ್ಟೆಲ್ಲಾ ವಿಸ್ತಾರವಾದ ಸಂಸ್ಕಾರಗಳ ಅಗತ್ಯವಿರುವುದಿಲ್ಲ. ವಾಮಮಾರ್ಗದಲ್ಲಿ ಕೇವಲ ಪ್ರಥಮ ನಾಲ್ಕು ಸಂಸ್ಕಾರಗಳನ್ನು — ಈಕ್ಷಣ, ಪ್ರೋಕ್ಷಣ, ತಾಡನ ಮತ್ತು ಅಭ್ಯುಕ್ಷಣಗಳನ್ನು ಮಾತ್ರವೇ ನೆರವೇರಿಸಿದರೆ ಸಾಕೆಂದು ಶಾಸ್ತ್ರವು ರಿಯಾಯಿತಿಯನ್ನು ನೀಡಿದೆ. ಅಕ್ಷಪಾಟನ ಸಂಸ್ಕಾರವು ಕುಂಡದಲ್ಲಿ ಚೇತನವನ್ನು ಜಾಗೃತಗೊಳಿಸಿ ಅದು ಹೋಮದ್ರವ್ಯಗಳನ್ನು ಸ್ವೀಕರಿಸಲು ಅರ್ಹವಾಗುವಂತೆ ಮಾಡುತ್ತದೆ. ವಾಮಮಾರ್ಗದಲ್ಲಿ ಸಂಕ್ಷಿಪ್ತ ಸಂಸ್ಕಾರಗಳನ್ನು ವಿಧಿಸಿರುವುದು ಆ ಮಾರ್ಗದ ಆಂತರಿಕ ಸಾಧನೆಯ ತೀವ್ರತೆಯನ್ನು ಸೂಚಿಸುತ್ತದೆ. ಶಾಸ್ತ್ರೀಯ ನಿಯಮಗಳನ್ನು ನಿಖರವಾಗಿ ಪಾಲಿಸುವುದೇ ಸಕಲ ಸಿದ್ಧಿಗಳ ರಹಸ್ಯವಾಗಿದೆ. ಆಚಾರ್ಯರ ಈ ಉಪದೇಶವನ್ನು ನಿಷ್ಠೆಯಿಂದ ಪಾಲಿಸುವುದೇ ಸಾಧಕನ ಆಧ್ಯಾತ್ಮಿಕ ಯಶಸ್ಸಿನ ಮೂಲ ಗುಟ್ಟಾಗಿದೆ. ಈ ನಿಯಮವು ವಾಮಮಾರ್ಗದ ವಿಶಿಷ್ಟ ಸಿದ್ಧಾಂತವೆನಿಸಿದೆ.

ವಿಶೇಷ ವಿವರಣೆ / ವ್ಯಾಖ್ಯಾನ:

ಅನುವಾದ: ಈ ವಿಧಿಯಿಂದ ಕುಂಡದ ಅಕ್ಷಪಾಟನ ಸಂಸ್ಕಾರವು ಸಿದ್ಧಿಸುತ್ತದೆ. ಆದರೆ ವಾಮಮಾರ್ಗದ ಹವನದಲ್ಲಿ ಕುಂಡಕ್ಕೆ ಕೇವಲ ಪ್ರಾರಂಭಿಕ ನಾಲ್ಕು ಸಂಸ್ಕಾರಗಳು ಮಾತ್ರವೇ ವಿಹಿತವಾಗಿವೆ.

ಅನುವಾದ:

ಈ ವಿಧಿಯಿಂದ ಕುಂಡದ ಅಕ್ಷಪಾಟನ ಸಂಸ್ಕಾರವು ನೆರವೇರುತ್ತದೆ; ಆದರೆ ವಾಮಮಾರ್ಗದ ಹೋಮದಲ್ಲಿ ಕುಂಡಕ್ಕೆ ಕೇವಲ ಮೊದಲ ನಾಲ್ಕು ಸಂಸ್ಕಾರಗಳು ಮಾತ್ರ ಶಾಸ್ತ್ರೋಕ್ತವಾಗಿವೆ.
