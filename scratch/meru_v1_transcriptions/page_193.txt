ಅಗ್ನುಪಸ್ಥಾನಮ್ (ಸಪ್ತಜಿಹ್ವಾವಿನ್ಯಾಸಃ)

ಶ್ಲೋಕ ೬೫-೬೭:

ಅಗ್ನಿಂ ಪ್ರಜ್ವಲಿತಂ ವನ್ದೇ ಜಾತವೇದಂ ಹುತಾಶನಮ್ ।
ಸುವರ್ಣವರ್ಣಮಮಲಂ ಸಮಿದ್ಧಂ ಸರ್ವತೋಮುಖಮ್ ॥೬೫॥
ಅನೇನ ಜ್ವಲಿತಂ ಮನ್ತ್ರೇಣೋಪತಿಷ್ಠೇದ್ಧುತಾಶನಮ್ ।
ಮುಖೇನೈವ ಧಮೇದಗ್ನಿಂ ಮುಖಾದೇಷ ವ್ಯಜಾಯತ ॥೬೬॥
ಕುಶೈಃ ಪಲಾಶಾದಿಪರ್ಣೈರ್ನಾಪವಿತ್ರೇಣ ಕೇನಚಿತ್ ।
ತತಃ ಸುವಿನ್ಯಸೇಜ್ಜೀಹ್ವಾ ದೇಹೇ ಮನ್ತ್ರೈರ್ವಿಭಾವಸೋಃ ॥೬೭॥

ಶಬ್ದಾರ್ಥ:

• ಅಗ್ನಿಮ್ — ಅಗ್ನಿದೇವನನ್ನು (ಪುಲ್ಲಿಂಗ, ದ್ವಿತೀಯಾ ಏಕವಚನ)
• ಪ್ರಜ್ವಲಿತಮ್ — ಚೆನ್ನಾಗಿ ಉರಿಯುತ್ತಿರುವ (ವಿಶೇಷಣ)
• ವನ್ದೇ — ನಮಸ್ಕರಿಸುತ್ತೇನೆ (ಲಟ್ ಲಕಾರ, ಉತ್ತಮ ಪುರುಷ ಏಕವಚನ)
• ಜಾತವೇದಮ್ — ಸರ್ವಜ್ಞನಾದ ಜಾತವೇದನನ್ನು (ದ್ವಿತೀಯಾ ಏಕವಚನ)
• ಹುತಾಶನಮ್ — ಹವಿಸ್ಸನ್ನು ಭಕ್ಷಿಸುವವನನ್ನು (ದ್ವಿತೀಯಾ ಏಕವಚನ)
• ಸುವರ್ಣವರ್ಣಮ್ — ಶುದ್ಧ ಬಂಗಾರದ ಬಣ್ಣವುಳ್ಳವನನ್ನು (ವಿಶೇಷಣ)
• ಅಮಲಮ್ — ಕಲ್ಮಷರಹಿತನಾದ ಪರಿಶುದ್ಧನನ್ನು (ವಿಶೇಷಣ)
• ಸಮಿದ್ಧಮ್ — ಸಮಿಧೆಗಳಿಂದ ಪ್ರಜ್ವಲಿಸುವವನನ್ನು (ವಿಶೇಷಣ)
• ಸರ್ವತೋಮುಖಮ್ — ಸಕಲ ದಿಕ್ಕುಗಳಲ್ಲಿಯೂ ಮುಖಗಳುಳ್ಳವನನ್ನು (ವಿಶೇಷಣ)
• ಅನೇನ — ಈ ಮೇಲೆ ಹೇಳಿದ (ತೃತೀಯಾ ಏಕವಚನ)
• ಜ್ವಲಿತಮ್ — ಪ್ರಜ್ವಲಿಸಿದ (ದ್ವಿತೀಯಾ ಏಕವಚನ)
• ಮನ್ತ್ರೇಣ — ಮಂತ್ರದಿಂದ (ತೃತೀಯಾ ಏಕವಚನ)
• ಉಪತಿಷ್ಠೇತ್ — ಉಪಸ್ಥಾನ ಮಾಡಬೇಕು, ಕೈಮುಗಿದು ಪ್ರಾರ್ಥಿಸಬೇಕು (ವಿಧಿಲಿಙ್ ಲಕಾರ)
• ಹುತಾಶನಮ್ — ಅಗ್ನಿದೇವನನ್ನು (ದ್ವಿತೀಯಾ ಏಕವಚನ)
• ಮುಖೇನ — ತನ್ನ ಸ್ವಂತ ಮುಖದಿಂದಲೇ (ತೃತೀಯಾ ಏಕವಚನ)
• ಏವ — ಮಾತ್ರವೇ (ಅವ್ಯಯ)
• ಧಮೇತ್ — ಗಾಳಿ ಊದಬೇಕು (ವಿಧಿಲಿಙ್ ಲಕಾರ)
• ಅಗ್ನಿಮ್ — ಅಗ್ನಿಗೆ (ದ್ವಿತೀಯಾ ಏಕವಚನ)
• ಮುಖಾತ್ — ವಿರಾಟ್ ಪುರುಷನ ಮುಖದಿಂದಲೇ (ಪಂಚಮೀ ಏಕವಚನ)
• ಏಷಃ — ಈ ಅಗ್ನಿಯು (ಪುಲ್ಲಿಂಗ, ಪ್ರಥಮಾ ಏಕವಚನ)
• ವ್ಯಜಾಯತ — ಜನ್ಮತಾಳಿದ್ದಾನೆ (ಲಙ್ ಲಕಾರ)
• ಕುಶೈಃ — ದರ್ಭೆಗಳಿಂದ (ತೃತೀಯಾ ಬಹುವಚನ)
• ಪಲಾಶಾದಿಪರ್ಣೈಃ — ಮುತ್ತುಗದ ಎಲೆ ಮೊದಲಾದವುಗಳಿಂದ (ತೃತೀಯಾ ಬಹುವಚನ)
• ನ — ಎಂದಿಗೂ ಮಾಡಬಾರದು (ನಿಷೇಧಾರ್ಥಕ ಅವ್ಯಯ)
• ಅಪವಿತ್ರೇಣ — ಅಶುದ್ಧವಾದ ಯಾವುದೇ ವಸ್ತುವಿನಿಂದ (ತೃತೀಯಾ ಏಕವಚನ)
• ಕೇನಚಿತ್ — ಯಾವುದರಿಂದಲೂ (ತೃತೀಯಾ ಏಕವಚನ)
• ತತಃ — ತರುವಾಯ (ಅವ್ಯಯ)
• ಸುವಿನ್ಯಸೇತ್ — ವಿಧಿವತ್ತಾಗಿ ವಿನ್ಯಾಸ ಮಾಡಬೇಕು (ವಿಧಿಲಿಙ್ ಲಕಾರ)
• ಜಿಹ್ವಾಃ — ಅಗ್ನಿಯ ಏಳು ನಾಲಿಗೆಗಳನ್ನು (ಸ್ತ್ರೀಲಿಂಗ, ದ್ವಿತೀಯಾ ಬಹುವಚನ)
• ದೇಹೇ — ಅಗ್ನಿದೇವನ ಶರೀರದಲ್ಲಿ (ಸಪ್ತಮೀ ಏಕವಚನ)
• ಮನ್ತ್ರೈಃ — ಶಾಸ್ತ್ರೋಕ್ತ ಮಂತ್ರಗಳಿಂದ (ತೃತೀಯಾ ಬಹುವಚನ)
• ವಿಭಾವಸೋಃ — ಪ್ರಕಾಶರೂಪಿಯಾದ ಅಗ್ನಿಯ (ಷಷ್ಠೀ ಏಕವಚನ)

ಭಾವಾರ್ಥ:

ಅಗ್ನಿಸ್ಥಾಪನೆಯ ನಂತರದ ಉಪಸ್ಥಾನ ಮತ್ತು ಜಿಹ್ವಾವಿನ್ಯಾಸದ ಪವಿತ್ರ ನಿಯಮಗಳನ್ನು ಈ ಶ್ಲೋಕಗಳಲ್ಲಿ ವಿವರಿಸಲಾಗಿದೆ. ಅಗ್ನಿಯು ಪ್ರಜ್ವಲಿಸಿದಾಗ ಸಾಧಕನು ಕೈಮುಗಿದು: 'ಅಗ್ನಿಂ ಪ್ರಜ್ವಲಿತಂ ವಂದೇ ಜಾತವೇದಂ ಹುತಾಶನಮ್ । ಸುವರ್ಣವರ್ಣಮಮಲಂ ಸಮಿದ್ಧಂ ಸರ್ವತೋಮುಖಮ್ ॥' ಎಂಬ ಮಂತ್ರದಿಂದ ಅಗ್ನಿದೇವನನ್ನು ವಂದಿಸಿ ಉಪಸ್ಥಾನ ಮಾಡಬೇಕು. ಅಗ್ನಿಯು ವಿರಾಟ್ ಪುರುಷನ ಮುಖದಿಂದ ಜನಿಸಿದವನಾದುದರಿಂದ, ಆತನನ್ನು ಪ್ರಜ್ವಲಿಸಲು ಆಚಾರ್ಯನು ತನ್ನ ಬಾಯಿಯ ಪವಿತ್ರ ಉಸಿರಿನಿಂದಾಗಲಿ, ಅಥವಾ ಕುಶ ಮತ್ತು ಪಲಾಶದ (ಮುತ್ತುಗದ) ಎಲೆಗಳಿಂದಾಗಲಿ ಮಾತ್ರ ಗಾಳಿ ಹಾಕಬೇಕು. ಯಾವುದೇ ಕಾರಣಕ್ಕೂ ಚರ್ಮದ ಬೀಸಣಿಗೆ, ಮುರಿದ ಬಟ್ಟೆ, ಅಥವಾ ಅಶುದ್ಧ ವಸ್ತುಗಳಿಂದ ಅಗ್ನಿಗೆ ಗಾಳಿ ಹಾಕಬಾರದು; ಹಾಗೆ ಮಾಡಿದರೆ ದೋಷವುಂಟಾಗುತ್ತದೆ. ತದನಂತರ ಅಗ್ನಿದೇವನ ಶರೀರದಲ್ಲಿ ಆತನ ಸಪ್ತಜಿಹ್ವೆಗಳನ್ನು (ಏಳು ನಾಲಿಗೆಗಳನ್ನು) ಶಾಸ್ತ್ರೋಕ್ತ ಮಂತ್ರಗಳಿಂದ ಧ್ಯಾನಿಸಿ ವಿನ್ಯಾಸ ಮಾಡಬೇಕು. ಅಗ್ನಿಯು ಕೇವಲ ಭೌತಿಕ ಬೆಂಕಿಯಲ್ಲ; ಆತನು ಸಾಕ್ಷಾತ್ ಜಾತವೇದನಾಗಿದ್ದು ಸಮಸ್ತ ಕರ್ಮಗಳ ಸಾಕ್ಷಿಯಾಗಿದ್ದಾನೆ. ಮುಖದ ಪವಿತ್ರ ಶ್ವಾಸದಿಂದ ಅಥವಾ ದರ್ಭೆಯಿಂದ ಮಾತ್ರ ಗಾಳಿ ಹಾಕುವುದು ಅಗ್ನಿಗೆ ಸಲ್ಲಿಸುವ ಪರಮ ಗೌರವವಾಗಿದೆ. ಯಾವುದೇ ಅಪವಿತ್ರ ವಸ್ತುವಿನ ಸಂಪರ್ಕವು ಯಾಗದ ಪಾವಿತ್ರ್ಯವನ್ನು ಕೆಡಿಸುವುದರಿಂದ ಆಚಾರ್ಯನು ಅತ್ಯಂತ ಜಾಗರೂಕನಾಗಿರಬೇಕು.

ವಿಶೇಷ ವಿವರಣೆ / ವ್ಯಾಖ್ಯಾನ:

ಅನುವಾದ: 'ಅಗ್ನಿಂ ಪ್ರಜ್ವಲಿತಂ ವಂದೇ ಜಾತವೇದಂ ಹುತಾಶನಮ್' ಮಂತ್ರದಿಂದ ಪ್ರಜ್ವಲಿತ ಅಗ್ನಿಯನ್ನು ಉಪಸ್ಥಾನ ಮಾಡಬೇಕು. ಅಗ್ನಿಯು ಮುಖದಿಂದ ಹುಟ್ಟಿದ್ದರಿಂದ ಮುಖದಿಂದಲೇ ಅಥವಾ ಕುಶ, ಪಲಾಶದ ಎಲೆಗಳಿಂದ ಗಾಳಿ ಹಾಕಬೇಕು; ಅಪವಿತ್ರ ವಸ್ತುಗಳಿಂದ ಎಂದಿಗೂ ಗಾಳಿ ಹಾಕಬಾರದು. ನಂತರ ಅಗ್ನಿಯ ಶರೀರದಲ್ಲಿ ಸಪ್ತಜಿಹ್ವೆಗಳ ವಿನ್ಯಾಸ ಮಾಡಬೇಕು.

ಅನುವಾದ:

'ಅಗ್ನಿಂ ಪ್ರಜ್ವಲಿತಂ ವಂದೇ ಜಾತವೇದಂ ಹುತಾಶನಮ್' ಮಂತ್ರದಿಂದ ಅಗ್ನಿಯನ್ನು ಉಪಸ್ಥಾನ ಮಾಡಬೇಕು. ಕುಶ ಅಥವಾ ಪಲಾಶ ಪರ್ಣಗಳಿಂದ ಮಾತ್ರವೇ ಗಾಳಿ ಹಾಕಿ ಪ್ರಜ್ವಲಿಸಬೇಕು; ಅಪವಿತ್ರ ವಸ್ತುಗಳಿಂದ ಗಾಳಿ ಹಾಕಬಾರದು. ನಂತರ ಅಗ್ನಿಯ ಶರೀರದಲ್ಲಿ ಸಪ್ತಜಿಹ್ವೆಗಳನ್ನು ವಿನ್ಯಾಸ ಮಾಡಬೇಕು.

ಶ್ಲೋಕ ೬೮-೭೦:

ಭಾಲಲಿಙ್ಗೋರುಮೂರ್ಧಾಸ್ಯನಾಸಾನೇತ್ರೇಷು ಚ ಕ್ರಮಾತ್ ।
ಸರ್ವಾಙ್ಗೇಷು ಚ ಜಿಹ್ವಾಸ್ತಾ ವಕ್ಷ್ಯನ್ತೇ ತ್ರಿವಿಧಾತ್ಮಿಕಾಃ ॥೬೮॥
ಹಿರಣ್ಯಾ ಕನಕಾ ರಕ್ತಾ ಕೃಷ್ಣಾ ಚೈವ ತು ಸುಪ್ರಭಾ ।
ಬಹುರೂಪಾ ಚಾತಿರಕ್ತಾ ಜಿಹ್ವಾಃ ಸಪ್ತೇತಿ ಸಾತ್ತ್ವಿಕಾಃ ॥೬೯॥
ಅನಲಾದಿಸಬಿನ್ದ್ವನ್ತಯಾದಿಸಾನ್ತಾಕ್ಷರಾನ್ವಿತಾಃ ।
ಮನ್ತ್ರೀ ಪ್ರವಿನ್ಯಸೇದ್ಧ್ವಯೋ ವಹ್ನೇರಙ್ಗಾನಿ ವೈ ಕ್ರಮಾತ್ ॥೭೦॥

ಶಬ್ದಾರ್ಥ:

• ಭಾಲ — ಹಣೆಯಲ್ಲಿ (ಲಲಾಟದಲ್ಲಿ) (ಸಪ್ತಮೀ ಏಕವಚನ)
• ಲಿಙ್ಗ — ಜನನೇಂದ್ರಿಯದಲ್ಲಿ (ಸಪ್ತಮೀ ಏಕವಚನ)
• ಊರು — ತೊಡೆಗಳಲ್ಲಿ (ಸಪ್ತಮೀ ದ್ವಿವಚನ)
• ಮೂರ್ಧ — ಶಿರಸ್ಸಿನಲ್ಲಿ (ಸಪ್ತಮೀ ಏಕವಚನ)
• ಆಸ್ಯ — ಮುಖದಲ್ಲಿ (ಸಪ್ತಮೀ ಏಕವಚನ)
• ನಾಸಾ — ನಾಸಿಕದಲ್ಲಿ (ಮೂಗಿನಲ್ಲಿ) (ಸಪ್ತಮೀ ಏಕವಚನ)
• ನೇತ್ರೇಷು — ಕಣ್ಣುಗಳಲ್ಲಿ (ಸಪ್ತಮೀ ಬಹುವಚನ)
• ಚ — ಮತ್ತು (ಅವ್ಯಯ)
• ಕ್ರಮಾತ್ — ಕ್ರಮವಾಗಿ (ಅವ್ಯಯ)
• ಸರ್ವಾಙ್ಗೇಷು — ಸಮಸ್ತ ಅಂಗಗಳಲ್ಲಿಯೂ (ಸಪ್ತಮೀ ಬಹುವಚನ)
• ಚ — ಮತ್ತು (ಅವ್ಯಯ)
• ಜಿಹ್ವಾಃ — ನಾಲಿಗೆಗಳು (ಸ್ತ್ರೀಲಿಂಗ, ಪ್ರಥಮಾ ಬಹುವಚನ)
• ತಾಃ — ಅವುಗಳು (ಸರ್ವನಾಮ)
• ವಕ್ಷ್ಯನ್ತೇ — ಹೇಳಲ್ಪಡುತ್ತವೆ (ಕರ್ಮಣಿ ಲಟ್)
• ತ್ರಿವಿಧಾತ್ಮಿಕಾಃ — ಸಾತ್ತ್ವಿಕ, ರಾಜಸ, ತಾಮಸ ಎಂಬ ತ್ರಿವಿಧ ಸ್ವರೂಪವುಳ್ಳವು (ಪ್ರಥಮಾ ಬಹುವಚನ)
• ಹಿರಣ್ಯಾ — ಹಿರಣ್ಯಾ ಎಂಬ ಪ್ರಥಮ ನಾಲಿಗೆಯು (ಪ್ರಥಮಾ ಏಕವಚನ)
• ಕನಕಾ — ಕನಕಾ ಎಂಬ ದ್ವಿತೀಯ ನಾಲಿಗೆಯು (ಪ್ರಥಮಾ ಏಕವಚನ)
• ರಕ್ತಾ — ರಕ್ತಾ ಎಂಬ ತೃತೀಯ ನಾಲಿಗೆಯು (ಪ್ರಥಮಾ ಏಕವಚನ)
• ಕೃಷ್ಣಾ — ಕೃಷ್ಣಾ ಎಂಬ ಚತುರ್ಥ ನಾಲಿಗೆಯು (ಪ್ರಥಮಾ ಏಕವಚನ)
• ಚ — ಮತ್ತು (ಅವ್ಯಯ)
• ಏವ — ಮಾತ್ರವೇ (ಅವ್ಯಯ)
• ತು — ಆದರೆ (ಅವ್ಯಯ)
• ಸುಪ್ರಭಾ — ಸುಪ್ರಭಾ ಎಂಬ ಪಂಚಮ ನಾಲಿಗೆಯು (ಪ್ರಥಮಾ ಏಕವಚನ)
• ಬಹುರೂಪಾ — ಬಹುರೂಪಾ ಎಂಬ ಷಷ್ಠ ನಾಲಿಗೆಯು (ಪ್ರಥಮಾ ಏಕವಚನ)
• ಚ — ಮತ್ತು (ಅವ್ಯಯ)
• ಅತಿರಕ್ತಾ — ಅತಿರಕ್ತಾ ಎಂಬ ಸಪ್ತಮ ನಾಲಿಗೆಯು (ಪ್ರಥಮಾ ಏಕವಚನ)
• ಜಿಹ್ವಾಃ — ನಾಲಿಗೆಗಳು (ಪ್ರಥಮಾ ಬಹುವಚನ)
• ಸಪ್ತ — ಏಳು ಸಂಖ್ಯೆಯುಳ್ಳವು (ಸಂಖ್ಯಾವಾಚಕ)
• ಇತಿ — ಎಂದು (ಅವ್ಯಯ)
• ಸಾತ್ತ್ವಿಕಾಃ — ಸಾತ್ತ್ವಿಕ ಜಿಹ್ವೆಗಳೆಂದು ಪ್ರಸಿದ್ಧವಾಗಿವೆ (ಪ್ರಥಮಾ ಬಹುವಚನ)
• ಅನಲಾದಿ — ರಂ ಎಂಬ ಅಗ್ನಿಬೀಜದಿಂದ ಪ್ರಾರಂಭಿಸಿ (ಅವ್ಯಯ)
• ಸಬಿನ್ದ್ವನ್ತ — ಅನುಸ್ವಾರದಿಂದ ಕೂಡಿದ (ವಿಶೇಷಣ)
• ಯಾದಿಸಾನ್ತಾಕ್ಷರಾನ್ವಿತಾಃ — ಯಂ, ರಂ, ಲಂ, ವಂ, ಶಂ, ಷಂ, ಸಂ ಎಂಬ ಸಪ್ತ ಬೀಜಾಕ್ಷರಗಳಿಂದ ಕೂಡಿದ (ಪ್ರಥಮಾ ಬಹುವಚನ)
• ಮನ್ತ್ರೀ — ಮಂತ್ರಜ್ಞನಾದ ಸಾಧಕನು (ಪುಲ್ಲಿಂಗ, ಪ್ರಥಮಾ ಏಕವಚನ)
• ಪ್ರವಿನ್ಯಸೇತ್ — ವಿನ್ಯಾಸ ಮಾಡಬೇಕು (ವಿಧಿಲಿಙ್ ಲಕಾರ)
• ಹೃದ್ಯಃ — ಪ್ರಸನ್ನಚಿತ್ತನಾಗಿ (ಪುಲ್ಲಿಂಗ, ಪ್ರಥಮಾ ಏಕವಚನ)
• ವಹ್ನೇಃ — ಅಗ್ನಿಯ (ಷಷ್ಠೀ ಏಕವಚನ)
• ಅಙ್ಗಾನಿ — ಅಂಗಗಳಲ್ಲಿ (ದ್ವಿತೀಯಾ ಬಹುವಚನ)
• ವೈ — ನಿಶ್ಚಯವಾಗಿ (ಅವ್ಯಯ)
• ಕ್ರಮಾತ್ — ಅನುಕ್ರಮವಾಗಿ (ಅವ್ಯಯ)

ಭಾವಾರ್ಥ:

ಅಗ್ನಿದೇವನ ಶರೀರದಲ್ಲಿ ಆತನ ಸಪ್ತಜಿಹ್ವೆಗಳ ಸ್ಥಾನ ಮತ್ತು ಬೀಜಾಕ್ಷರ ವಿನ್ಯಾಸವನ್ನು ಈ ಶ್ಲೋಕಗಳಲ್ಲಿ ನಿರೂಪಿಸಲಾಗಿದೆ. ಅಗ್ನಿಯ ಲಲಾಟ, ಜನನಾಂಗ, ತೊಡೆಗಳು, ಶಿರಸ್ಸು, ಮುಖ, ನಾಸಿಕ ಮತ್ತು ನೇತ್ರಗಳಲ್ಲಿ ಕ್ರಮವಾಗಿ ಏಳು ಸಾತ್ತ್ವಿಕ ಜಿಹ್ವೆಗಳು ನೆಲೆಸಿವೆ: ೧. ಹಿರಣ್ಯಾ, ೨. ಕನಕಾ, ೩. ರಕ್ತಾ, ೪. ಕೃಷ್ಣಾ, ೫. ಸುಪ್ರಭಾ, ೬. ಬಹುರೂಪಾ ಹಾಗೂ ೭. ಅತಿರಕ್ತಾ. ಸಾಧಕನು ಯಂ, ರಂ, ಲಂ, ವಂ, ಶಂ, ಷಂ, ಸಂ ಎಂಬ ಸಪ್ತ ಅನುಸ್ವಾರಯುಕ್ತ ಬೀಜಾಕ್ಷರಗಳಿಂದ ಅಗ್ನಿಯ ಆಯಾ ಅಂಗಗಳಲ್ಲಿ ಈ ಸಪ್ತಜಿಹ್ವೆಗಳನ್ನು ವಿನ್ಯಾಸ ಮಾಡಬೇಕು. ಈ ಜಿಹ್ವೆಗಳೇ ಅಗ್ನಿಯಲ್ಲಿ ಅರ್ಪಿತವಾದ ವಿವಿಧ ಹವಿರ್ದ್ರವ್ಯಗಳನ್ನು ಸ್ವೀಕರಿಸಿ ಆಯಾ ದೇವತೆಗಳಿಗೆ ತಲುಪಿಸುವ ಸಾಧನಗಳಾಗಿವೆ. ಇವುಗಳಲ್ಲಿ ಶಾಂತಿಕ ಕರ್ಮಕ್ಕೆ ಹಿರಣ್ಯಾ, ಪೌಷ್ಟಿಕ ಕರ್ಮಕ್ಕೆ ಕನಕಾ, ವಶೀಕರಣಕ್ಕೆ ರಕ್ತಾ, ಉಚ್ಚಾಟನಕ್ಕೆ ಸುಪ್ರಭಾ ಮುಂತಾದವು ಪ್ರಶಸ್ತವಾಗಿವೆ. ಸಪ್ತಜಿಹ್ವೆಗಳು ಅಗ್ನಿಯ ಏಳು ಶಕ್ತಿಗಳಾಗಿದ್ದು, ಆಯಾ ಕರ್ಮಗಳಿಗೆ ತಕ್ಕಂತೆ ಹವಿಸ್ಸನ್ನು ಸ್ವೀಕರಿಸಿ ಫಲ ನೀಡುತ್ತವೆ. ಯಂ, ರಂ, ಲಂ, ವಂ ಮುಂತಾದ ಬೀಜಾಕ್ಷರಗಳು ಪಂಚಭೂತಗಳು ಮತ್ತು ಮನಸ್ಸುಗಳನ್ನು ಅಗ್ನಿಯಲ್ಲಿ ಏಕೀಕರಿಸುತ್ತವೆ. ಈ ಜಿಹ್ವಾವಿನ್ಯಾಸದಿಂದ ಅಗ್ನಿಯು ಸಮಸ್ತ ದೇವತೆಗಳ ಮುಖವಾಗಿ ಪೂರ್ಣ ಸನ್ನದ್ಧನಾಗುತ್ತಾನೆ.

ವಿಶೇಷ ವಿವರಣೆ / ವ್ಯಾಖ್ಯಾನ:

ಅನುವಾದ: ಅಗ್ನಿಯ ಹಣೆ, ಲಿಂಗ, ತೊಡೆ, ಶಿರ, ಮುಖ, ನಾಸಿಕ, ನೇತ್ರಗಳಲ್ಲಿ ಸಪ್ತಜಿಹ್ವೆಗಳು ನೆಲೆಸಿವೆ. ಅವು ಹಿರಣ್ಯಾ, ಕನಕಾ, ರಕ್ತಾ, ಕೃಷ್ಣಾ, ಸುಪ್ರಭಾ, ಬಹುರೂಪಾ ಮತ್ತು ಅತಿರಕ್ತಾ. ಯಂ, ರಂ, ಲಂ, ವಂ, ಶಂ, ಷಂ, ಸಂ ಎಂಬ ಬೀಜಗಳಿಂದ ಅಗ್ನಿಯ ಆಯಾ ಅಂಗಗಳಲ್ಲಿ ಜಿಹ್ವೆಗಳನ್ನು ವಿನ್ಯಾಸ ಮಾಡಬೇಕು.

ಅನುವಾದ:

ಅಗ್ನಿಯ ಲಲಾಟಾದಿ ಅಂಗಗಳಲ್ಲಿ ಹಿರಣ್ಯಾ, ಕನಕಾ, ರಕ್ತಾ, ಕೃಷ್ಣಾ, ಸುಪ್ರಭಾ, ಬಹುರೂಪಾ ಮತ್ತು ಅತಿರಕ್ತಾ ಎಂಬ ಸಪ್ತಜಿಹ್ವೆಗಳನ್ನು ಯಂ ರಂ ಲಂ ವಂ ಶಂ ಷಂ ಸಂ ಬೀಜಗಳಿಂದ ಕ್ರಮವಾಗಿ ವಿನ್ಯಾಸ ಮಾಡಬೇಕು.

ಶ್ಲೋಕ ೭೧-೭೩:

ಸಹಸ್ರಾರ್ಚಿಃ ಸ್ವಸ್ತಿಪೂರ್ಣ ಉತ್ತಿಷ್ಠಪುರುಷಸ್ತಥಾ ।
ಧೂಮವ್ಯಾಪೀ ಸಪ್ತಜಿಹ್ವೋ ಧನುರ್ಧರ ಇತೀರಿತಾಃ ॥೭೧॥
ಜಾತವೇದಾಃ ಸಪ್ತಮಶ್ಚ ಹವ್ಯವಾಹಸ್ತಥಾಷ್ಟಮಃ ।
ಅಗ್ನೇರ್ನಾಮಾನಿ ಚೈತಾನಿ ಸಪ್ತಾರ್ದ್ವೇ ತು ವಿಚಿನ್ತಯೇತ್ ॥೭೨॥
ಪ್ರಾಣಾಪಾನೌ ವ್ಯಾನೋದಾನೌ ಸಮಾನಶ್ಚೈವ ಪಞ್ಚಮಃ ।
ನಾಗಃ ಕೂರ್ಮೋಽಥ ಕೃಕರೋ ದೇವದತ್ತೋ ಧನಞ್ಜಯಃ ॥೭೩॥

ಶಬ್ದಾರ್ಥ:

• ಸಹಸ್ರಾರ್ಚಿಃ — ಸಹಸ್ರಾರು ಜ್ವಾಲೆಗಳುಳ್ಳವನು (ಪ್ರಥಮಾ ಏಕವಚನ)
• ಸ್ವಸ್ತಿಪೂರ್ಣಃ — ಮಂಗಲಪೂರ್ಣನಾದವನು (ಪ್ರಥಮಾ ಏಕವಚನ)
• ಉತ್ತಿಷ್ಠಪುರುಷಃ — ಉನ್ನತ ತೇಜಸ್ಸಿನಿಂದ ಎದ್ದುನಿಂತ ಪುರುಷನು (ಪ್ರಥಮಾ ಏಕವಚನ)
• ತಥಾ — ಹಾಗೆಯೇ (ಅವ್ಯಯ)
• ಧೂಮವ್ಯಾಪೀ — ಸಕಲ ದಿಕ್ಕುಗಳಲ್ಲಿ ಧೂಮವನ್ನು ಹರಡುವವನು (ಪ್ರಥಮಾ ಏಕವಚನ)
• ಸಪ್ತಜಿಹ್ವಃ — ಏಳು ನಾಲಿಗೆಗಳುಳ್ಳವನು (ಪ್ರಥಮಾ ಏಕವಚನ)
• ಧನುರ್ಧರಃ — ಧನುಸ್ಸನ್ನು ಧರಿಸಿದವನು (ಪ್ರಥಮಾ ಏಕವಚನ)
• ಇತಿ — ಎಂದು (ಅವ್ಯಯ)
• ಈರಿತಾಃ — ಹೇಳಲ್ಪಟ್ಟಿದ್ದಾರೆ (ಪ್ರಥಮಾ ಬಹುವಚನ)
• ಜಾತವೇದಾಃ — ಸರ್ವಜ್ಞನಾದ ಜಾತವೇದನು, ಏಳನೆಯವನು (ಪ್ರಥಮಾ ಏಕವಚನ)
• ಸಪ್ತಮಃ — ಏಳನೆಯ ನಾಮವು (ಪ್ರಥಮಾ ಏಕವಚನ)
• ಚ — ಮತ್ತು (ಅವ್ಯಯ)
• ಹವ್ಯವಾಹಃ — ಹವಿಸ್ಸನ್ನು ದೇವತೆಗಳಿಗೆ ಒಯ್ಯುವವನು (ಪ್ರಥಮಾ ಏಕವಚನ)
• ತಥಾ — ಹಾಗೆಯೇ (ಅವ್ಯಯ)
• ಅಷ್ಟಮಃ — ಎಂಟನೆಯ ನಾಮವು (ಪ್ರಥಮಾ ಏಕವಚನ)
• ಅಗ್ನೇಃ — ಅಗ್ನಿದೇವನ (ಷಷ್ಠೀ ಏಕವಚನ)
• ನಾಮಾನಿ — ಅಷ್ಟನಾಮಗಳು (ದ್ವಿತೀಯಾ ಬಹುವಚನ)
• ಚ — ಮತ್ತು (ಅವ್ಯಯ)
• ಏತಾನಿ — ಇವುಗಳನ್ನು (ದ್ವಿತೀಯಾ ಬಹುವಚನ)
• ಸಪ್ತಾರ್ದ್ವೇ — ಅಗ್ನಿಯ ಶರೀರದಲ್ಲಿ (ಸಪ್ತಮೀ ಏಕವಚನ)
• ತು — ಆದರೆ (ಅವ್ಯಯ)
• ವಿಚಿನ್ತಯೇತ್ — ಧ್ಯಾನಿಸಬೇಕು (ವಿಧಿಲಿಙ್ ಲಕಾರ)
• ಪ್ರಾಣಾಪಾನೌ — ಪ್ರಾಣ ಮತ್ತು ಅಪಾನ ವಾಯುಗಳು (ಪ್ರಥಮಾ ದ್ವಿವಚನ)
• ವ್ಯಾನೋದಾನೌ — ವ್ಯಾನ ಮತ್ತು ಉದಾನ ವಾಯುಗಳು (ಪ್ರಥಮಾ ದ್ವಿವಚನ)
• ಸಮಾನಃ — ಸಮಾನ ವಾಯುವು (ಪ್ರಥಮಾ ಏಕವಚನ)
• ಚ — ಮತ್ತು (ಅವ್ಯಯ)
• ಏವ — ಮಾತ್ರವೇ (ಅವ್ಯಯ)
• ಪಞ್ಚಮಃ — ಐದನೆಯ ಪ್ರಾಣವಾಯುವು (ಪ್ರಥಮಾ ಏಕವಚನ)
• ನಾಗಃ — ನಾಗ ಎಂಬ ಉಪಪ್ರಾಣವು (ಪ್ರಥಮಾ ಏಕವಚನ)
• ಕೂರ್ಮಃ — ಕೂರ್ಮ ಎಂಬ ಉಪಪ್ರಾಣವು (ಪ್ರಥಮಾ ಏಕವಚನ)
• ಅಥ — ನಂತರ (ಅವ್ಯಯ)
• ಕೃಕರಃ — ಕೃಕರ ಎಂಬ ಉಪಪ್ರಾಣವು (ಪ್ರಥಮಾ ಏಕವಚನ)
• ದೇವದತ್ತಃ — ದೇವದತ್ತ ಎಂಬ ಉಪಪ್ರಾಣವು (ಪ್ರಥಮಾ ಏಕವಚನ)
• ಧನಞ್ಜಯಃ — ಧನಂಜಯ ಎಂಬ ಉಪಪ್ರಾಣವು (ಪ್ರಥಮಾ ಏಕವಚನ)

ಭಾವಾರ್ಥ:

ಅಗ್ನಿಯ ಅಷ್ಟಮೂರ್ತಿಗಳು ಹಾಗೂ ದಶಪ್ರಾಣಗಳ ಆವಾಹನೆಯನ್ನು ಈ ಶ್ಲೋಕಗಳಲ್ಲಿ ನಿರೂಪಿಸಲಾಗಿದೆ. ಅಗ್ನಿಯ ಎಂಟು ದಿವ್ಯ ನಾಮಗಳೆಂದರೆ: ೧. ಸಹಸ್ರಾರ್ಚಿ, ೨. ಸ್ವಸ್ತಿಪೂರ್ಣ, ೩. ಉತ್ತಿಷ್ಠಪುರುಷ, ೪. ಧೂಮವ್ಯಾಪೀ, ೫. ಸಪ್ತಜಿಹ್ವ, ೬. ಧನುರ್ಧರ, ೭. ಜಾತವೇದ ಹಾಗೂ ೮. ಹವ್ಯವಾಹ. ಈ ಎಂಟು ನಾಮಗಳಿಂದ ಅಗ್ನಿಯ ಅಷ್ಟಮೂರ್ತಿಗಳನ್ನು ಆತನ ಶರೀರದಲ್ಲಿ ಪ್ರದಕ್ಷಿಣವಾಗಿ ಧ್ಯಾನಿಸಬೇಕು. ತದನಂತರ ಮಾನವ ಶರೀರದಲ್ಲಿರುವಂತೆಯೇ ಅಗ್ನಿದೇವನ ದಿವ್ಯ ಶರೀರದಲ್ಲಿಯೂ ಪಂಚ ಮುಖ್ಯ ಪ್ರಾಣಗಳಾದ ಪ್ರಾಣ, ಅಪಾನ, ವ್ಯಾನ, ಉದಾನ, ಸಮಾನ ಹಾಗೂ ಪಂಚ ಉಪಪ್ರಾಣಗಳಾದ ನಾಗ, ಕೂರ್ಮ, ಕೃಕರ, ದೇವದತ್ತ ಮತ್ತು ಧನಂಜಯ ಎಂಬ ಹತ್ತು ಪ್ರಾಣಶಕ್ತಿಗಳನ್ನು ಆವಾಹಿಸಬೇಕು. ಈ ಪ್ರಾಣಪ್ರತಿಷ್ಠೆಯು ಯಜ್ಞಾಗ್ನಿಯನ್ನು ಸಾಕ್ಷಾತ್ ಜೀವಂತ ಬ್ರಹ್ಮಾಂಡ ಪುರುಷನನ್ನಾಗಿ ಪರಿವರ್ತಿಸುತ್ತದೆ. ಅಗ್ನಿಯ ಅಷ್ಟಮೂರ್ತಿಗಳು ಅಷ್ಟದಿಕ್ಕುಗಳ ರಕ್ಷಣೆಯನ್ನು ಖಚಿತಪಡಿಸಿ ಯಾಗಕ್ಷೇತ್ರವನ್ನು ದೈವೀ ದುರ್ಗವನ್ನಾಗಿ ಮಾಡುತ್ತವೆ. ದಶಪ್ರಾಣಗಳ ಆವಾಹನೆಯು ಅಗ್ನಿಯಲ್ಲಿ ಜೀವಂತ ಪ್ರಾಣಶಕ್ತಿಯನ್ನು ತುಂಬಿ ಆತನನ್ನು ಚೈತನ್ಯಮಯನನ್ನಾಗಿ ಮಾಡುತ್ತದೆ. ಈ ಸಮಗ್ರ ಧ್ಯಾನವು ಸಾಧಕನಿಗೆ ದೀರ್ಘಾಯುಷ್ಯ, ತೇಜಸ್ಸು ಮತ್ತು ಆಧ್ಯಾತ್ಮಿಕ ಬಲವನ್ನು ನೀಡುತ್ತದೆ.

ವಿಶೇಷ ವಿವರಣೆ / ವ್ಯಾಖ್ಯಾನ:

ಅನುವಾದ: ಅಗ್ನಿಯ ಎಂಟು ಹೆಸರುಗಳು — ಸಹಸ್ರಾರ್ಚಿ, ಸ್ವಸ್ತಿಪೂರ್ಣ, ಉತ್ತಿಷ್ಠಪುರುಷ, ಧೂಮವ್ಯಾಪೀ, ಸಪ್ತಜಿಹ್ವ, ಧನುರ್ಧರ, ಜಾತವೇದಾ ಮತ್ತು ಹವ್ಯವಾಹ. ಇವುಗಳನ್ನು ಅಗ್ನಿಯ ಶರೀರದಲ್ಲಿ ಧ್ಯಾನಿಸಬೇಕು. ನಂತರ ಪ್ರಾಣ, ಅಪಾನ, ವ್ಯಾನ, ಉದಾನ, ಸಮಾನ, ನಾಗ, ಕೂರ್ಮ, ಕೃಕರ, ದೇವದತ್ತ ಮತ್ತು ಧನಂಜಯ ಎಂಬ ದಶವಾಯುಗಳನ್ನು ಅಗ್ನಿಯಲ್ಲಿ ಧ್ಯಾನಿಸಬೇಕು.

ಅನುವಾದ:

ಅಗ್ನಿಯ ಅಷ್ಟನಾಮಗಳಾದ ಸಹಸ್ರಾರ್ಚಿ, ಸ್ವಸ್ತಿಪೂರ್ಣ, ಉತ್ತಿಷ್ಠಪುರುಷ, ಧೂಮವ್ಯಾಪೀ, ಸಪ್ತಜಿಹ್ವ, ಧನುರ್ಧರ, ಜಾತವೇದಾ ಮತ್ತು ಹವ್ಯವಾಹಗಳನ್ನು ಹಾಗೂ ದಶಪ್ರಾಣವಾಯುಗಳನ್ನು ಅಗ್ನಿಯ ಶರೀರದಲ್ಲಿ ಧ್ಯಾನಿಸಬೇಕು.
